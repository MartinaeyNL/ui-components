import {Markdown} from "@storybook/blocks";

export function getNPMVersionBadge(packageJson) {
    return <a href={`https://npmjs.com/package/${packageJson.name}`}>
        <img alt="NPM version" src={`https://badgen.net/badge/npm/v${packageJson.version}?scale=1.2&color=orange`}></img>
    </a>
}

export function getLicenseBadge(packageJson) {
    return <a href={packageJson.repository}>
        <img alt="License" src={`https://badgen.net/badge/license/${packageJson.license}?scale=1.2&color=blue`}></img>
    </a>
}

export function getAccessibilityBadge(type) {
    let url;
    let text;
    switch (type) {
        case "vaadin-WCAG2.1-AA":
            text = "WCAG%202.1%20-%20Level%20AA"
            url = "https://vaadin.com/accessibility"
    }
    return <a href={url}>
        <img alt="Accessibility" src={`https://badgen.net/badge/Accessibility/${text}?scale=1.2&color=green`}></img>
    </a>
}

export function getDescription(docs) {
    return docs.description ? <Markdown id="description">{docs.description}</Markdown> : <p id="description"><i>The description for {docs.tagName || "this component"} could not be found.</i></p>
}

export function getCollapsibleElement(index, title, titleSuffix, content) {
    return <div className="details-item">
        <input type="checkbox" id={"details-" + index} />
        <label htmlFor={"details-" + index}>
            <span>{title}</span>
            {titleSuffix}
        </label>
        <div className="details-content">
            {content}
        </div>
    </div>
}

export function getBundleReportCollapsible(index, badge, content) {
    return getCollapsibleElement(index, "Bundle report", badge, content)
}
